## 可复现计算脚本

在仓库根目录使用项目虚拟环境运行；脚本只读输入并写 stdout。脚本共 31 行。

```python
from pathlib import Path
from statistics import NormalDist
import numpy as np
d = Path("docs/research_cycles/V80-NBLDPC-JAN21")
g, pbz = np.load(d/"gamma_f03.npz"), np.load(d/"gamma_f03_pb.npz")
qinv = lambda e: NormalDist().inv_cdf(1-e)
h_corr = {"1M": .8012690084416184, "1p5M": .8272902027770036, "2M": .8333327179427281}
ms, anchor = {"1M":197, "1p5M":203, "2M":204}, .83256272
for s in ("1M", "1p5M", "2M"):
    g1, g2 = np.asarray(g[f"{s}_gamma1_L1"],float), np.asarray(g[f"{s}_gamma2_L2condU1"],float)
    pb = np.asarray(pbz[f"{s}_p_b"],float); p = g1[:,None,:]*g2; j = p*pb[None,None,:]
    z = np.zeros_like(p); m = p>0; z[m] = -np.log2(p[m])
    H = float(np.sum(j*z)); V = float(np.sum(j*z*z)-H*H)
    p2 = p.sum(axis=0); z2 = np.zeros_like(p2); m2 = p2>0; z2[m2] = -np.log2(p2[m2])
    Hu2 = float(np.sum(p2*pb[None,:]*z2))
    c = np.divide(p,p2[None,:,:],out=np.zeros_like(p),where=p2[None,:,:]>0)
    zc = np.zeros_like(c); mc = c>0; zc[mc] = -np.log2(c[mc])
    Hu1u2 = float(np.sum(j*zc)); Hparts = float(g[f"{s}_H_L1"]+g[f"{s}_H_L2"])
    print("R1",s,"H,V,sqrtV,HU2|B,HU1|U2B,Hparts=",H,V,V**.5,Hu2,Hu1u2,Hparts)
    for n in (1024,2048,4096,16384):
        for e in ((.01,.001) if n<=2048 else (.01,)):
            L = n*H+(n*V)**.5*qinv(e); f = L/(n*H); tag = 64/(n*H); w = .5*np.log2(n)/(n*H)
            print("R2",s,n,e,"f,ftag,interval,tag_interval=",f,f+tag,(f-w,f+w),(f+tag-w,f+tag+w))
    for label,h in (("NPZ",H),("prompt_H",h_corr[s]),("anchor",anchor)):
        fm = 5*ms[s]/(1024*h); fs = 1+(V/1024)**.5*qinv(.01)/h; tg = 64/(1024*h)
        print("R3",s,label,"fmeas,fstar,finite,code_gap,tag,total=",fm,fs,fs-1,fm-fs,tg,fm+tg)
        f1 = 1+(V/1024)**.5*qinv(.01)/h; f2 = 1+(V/2048)**.5*qinv(.01)/h
        t1,t2 = 64/(1024*h),64/(2048*h)
        print("R4",s,label,"finite_save,tag_save,total=",f1-f2,t1-t2,f1-f2+t1-t2)
print("R5 observed_rates,gate=",2/240,3/240,12/240)
g.close(); pbz.close()
```
